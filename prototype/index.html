B. Critical ingestion flow

docs/architecture/ingestion-flow.mmd
%% B. Critical ingestion flow with decision branches (proposed).
flowchart TD
  A([Operator selects source]) --> B{Source enabled<br/>and config valid?}
  B -- no --> B1[SOURCE_DISABLED or config error<br/>refuse + audit] --> Z([End: no run])
  B -- yes --> C{Mapping valid?<br/>required fields mapped}
  C -- no --> C1[FIELD_MAPPING_FAILED<br/>run BLOCKED] --> Z
  C -- yes --> D{Mode}
  D -- API --> E[Get token from vault<br/>authenticate]
  E --> E1{Auth ok?}
  E1 -- no --> E2[AUTHENTICATION_FAILED<br/>run FAILED] --> Z
  E1 -- yes --> F[Fetch page with watermark]
  F --> F1{HTTP status}
  F1 -- "429 / timeout" --> F2[Backoff using Retry-After<br/>retry page] --> F
  F1 -- "200" --> F3{More pages?}
  F3 -- yes --> F
  F3 -- no --> G
  D -- Import --> H[Upload to quarantine<br/>scan type, size, signature]
  H --> H1{File safe?}
  H1 -- no --> H2[FILE_FORMAT_INVALID<br/>quarantined, not parsed] --> Z
  H1 -- yes --> H3[Parse values only<br/>flag formula cells] --> H4[Dry run + approval] --> G
  G{Schema matches<br/>registered version?}
  G -- no --> G1[SOURCE or FILE_SCHEMA_MISMATCH<br/>stop per policy] --> Z
  G -- yes --> I[Map + transform<br/>mapping vN]
  I --> J{Record valid?}
  J -- no --> J1[Reject with field reasons<br/>error queue] --> R
  J -- yes --> K{Duplicate or match?}
  K -- "in-batch duplicate" --> K1[DUPLICATE_RECORD<br/>held back] --> R
  K -- "cross-source match" --> K2[MATCH_REQUIRES_REVIEW<br/>manual review] --> K3{Reviewer decision}
  K3 -- "same person" --> R
  K3 -- "different person" --> L
  K -- "unique or update" --> L[Deliver to MASTER_SR1<br/>with idempotency key]
  L --> M{SR1 response}
  M -- "2xx accepted" --> M1[Poll commit status] --> N
  M -- "5xx transient" --> M2{Attempts below max?}
  M2 -- yes --> M3[Schedule retry<br/>exp. backoff + jitter] --> L
  M2 -- no --> M4[Retry exhausted<br/>dead letter] --> R
  M -- "4xx permanent" --> M5[MASTER_WRITE_FAILED<br/>dead letter, fix at source] --> R
  M -- "timeout" --> M6[MASTER_WRITE_OUTCOME_UNKNOWN]
  M6 --> M7[Reconcile: query by idempotency key]
  M7 -- found --> N
  M7 -- "not found" --> L
  N[Written to SR1] --> O{Visible on SR2<br/>within threshold?}
  O -- "not yet" --> O1[REPLICA_SYNC_PENDING<br/>lag alert] --> O
  O -- "window expired" --> O2[REPLICA_SYNC_FAILED<br/>request re-sync] --> O
  O -- yes --> P[Confirmed on SR2]
  P --> R[Reconcile run<br/>received = rejected + held + accepted<br/>accepted = written + failed + retry + unknown]
  R --> S([Run status derived from record states])

    +list() Source[]

  }
  class SourceConfigurationService {
    +validate(cfg) Error[]
    +checkOutboundUrl(url) Verdict
    +resolveSecret(ref) Credential
  }

    +authenticate(source)
    +fetchPage(cursor, watermark) Page
    -handleThrottle(retryAfter)
    -validateResponseSchema(page)
  }
  class FileImportAdapter {
    +receive(upload) QuarantinedFile
    +parse(file) RowStream
  }
  class UploadValidationService {
    +checkExtension()
    +checkSignature()
    +checkSize()
    +sanitizeName()
    +scanMalware()
  }
  class SchemaRegistryService {
    +register(sourceId, schema) Version
    +fingerprint(headers) string
    +compare(incoming, registered) Drift
  }
  class MappingService {
    +validate(mapping) MappingReport
    +publish(mapping, note) Version
    +test(mapping, sample) Preview
  }
  class TransformationService {
    +apply(rule, values) value
  }
  class ValidationService {
    +validateCanonical(record) Issue[]
  }
  class RecordMatchingService {
    +inBatch(records) Duplicate[]
    +againstMaster(record) MatchCandidate
  }
  class IngestionOrchestrator {
    +start(sourceId, mode, opts) Run
    +advance(runId)
    +checkpoint(runId, stage, offset)
  }
  class IngestionRunService {
    +status(runId) RunStatus
    +counts(runId) StageCounts
  }
  class DeliveryService {
    +deliver(batch)
    +idempotencyKey(record) string
  }
  class MasterRegistryAdapter {
    +upsert(record, key) Ack
    +findByKey(key) MasterRef
    +commitStatus(ingestId) CommitState
  }
  class ReplicaSyncMonitor {
    +probe(internalRecordId) bool
    +lag() Duration
  }
  class RetryScheduler {
    +schedule(record, attempt) Instant
    +due() Record[]
  }
  class DeadLetterService {
    +park(record, reason)
    +release(recordId)
  }
  class ReconciliationService {
    +reconcile(runId) ReconciliationResult
    +resolveUnknown(record)
  }
  class AuditService {

  UI->>Orch: POST /runs
  Orch->>Aud: RUN_STARTED
  Orch->>V: get credential by reference
  V-->>Orch: short-lived token
  Orch->>Ad: extract(watermark)
  loop each page
    Ad->>Src: GET /records?cursor and updated_since
    alt 200 OK
      Src-->>Ad: page + next cursor
    else 429 Too Many Requests
      Src-->>Ad: Retry-After 2s
      Ad->>Ad: wait, retry same page
    else timeout
      Ad->>Ad: backoff, retry same page
    else 401
      Src-->>Ad: invalid_client
      Ad-->>Orch: AUTHENTICATION_FAILED
      Orch->>Aud: run FAILED
    end
  end
  Ad-->>Orch: raw records
  Orch->>MV: schema check, map vN, validate, de-duplicate
  MV-->>Orch: valid / invalid / duplicate / review
  Orch->>Q: enqueue delivery jobs (batches of 10)
  Q->>Del: job
  Del->>SR1: upsert(record, idempotency key)
  alt 2xx accepted
    SR1-->>Del: ingest_id
    Del->>SR1: poll commit status
    SR1-->>Del: committed, internal_record_id
  else 503 transient
    SR1-->>Del: 503
    Del->>Q: schedule retry (exp. backoff + jitter)
    Q->>Del: retry later with same key
  else 422 permanent
    SR1-->>Del: 422
    Del->>Q: dead letter
  else timeout, outcome unknown
    Del-->>Orch: MASTER_WRITE_OUTCOME_UNKNOWN
    Rec->>SR1: query by idempotency key
    alt found
      SR1-->>Rec: internal_record_id
      Rec->>Orch: mark WRITTEN, no resend
    else not found
      Rec->>Q: requeue with same key (safe)
    end
  end
  SR1-)SR2: replication (async)
  Del->>SR2: read probe by internal_record_id
  alt visible
    SR2-->>Del: found
  else lag above threshold
    Del->>Aud: REPLICA_SYNC_PENDING alert
  end
  Orch->>Rec: reconcile run
  Rec-->>Orch: buckets + identity checks
  Orch->>Aud: run status, metrics
  UI-->>Op: run details, errors, reconciliation

%% F. File import sequence: quarantine, validation, preview, dry run, approval, batch processing.
sequenceDiagram
  autonumber
  actor DO as Desk officer or operator
  participant UI as Console
  participant API as DataBridge API
  participant OS as Object store
  participant UV as Upload validation
  participant P as Parser (values only)
  participant SR as Schema registry
  participant M as Mapping, validation, dedup
  participant Del as Delivery service
  participant SR1 as MASTER_SR1
  participant SR2 as MASTER_SR2
  participant Rec as Reconciliation
  participant Aud as Audit

  DO->>UI: choose file (.csv or .xlsx)
  UI->>API: upload (pre-signed, multipart)
  API->>OS: write to quarantine zone
  API->>UV: scan
  UV->>UV: sanitise name, extension allow-list, size, signature, AV scan
  alt rejected
    UV-->>API: FILE_FORMAT_INVALID
    API->>Aud: upload rejected, file kept in quarantine
    API-->>UI: reasons (file never parsed)
  else passed
    UV->>OS: move to staging, record sha256
    API->>P: parse (no formulas, no macros)
    P-->>API: headers + rows, formula cells flagged
    API->>SR: compare header fingerprint with template version
    alt mismatch
      SR-->>API: FILE_SCHEMA_MISMATCH (missing and unexpected columns)
      API-->>UI: stop, show differences
    else match
      API-->>UI: preview first rows (sensitive columns masked)
      DO->>UI: run dry run
      UI->>API: dry run
      API->>M: map, validate, de-duplicate (no delivery)
      M-->>API: would deliver / invalid / duplicate / review
      API-->>UI: dry-run report
      DO->>UI: approve
      UI->>API: start run (approvedBy, dryRunRef)
      API->>Aud: RUN_STARTED with approval
      loop each batch of 10
        API->>M: map + validate + dedup
        M->>Del: deliverable records
        Del->>SR1: upsert with idempotency key
        SR1-->>Del: ack, 5xx, 422 or timeout
      end
      Del->>OS: write redacted reject file to reject zone
      SR1-)SR2: replication
      Del->>SR2: read probe
      API->>Rec: reconcile
      Rec-->>UI: received vs rejected vs written vs replica
      DO->>UI: download error report (CSV, injection-safe)
    end
  end
